#include <stdio.h>

int main() {

    // Arreglo que almacena 5 calificaciones de tipo float

    float suma = 0;
    float promedio;

    // Captura de calificaciones
    // cada numero introducido por el usuario se guarda en un espacio del arreglo calificaciones

        printf("Calificacion de materia %d: ", i + 1);
        scanf("%f", &calificaciones[i]);
        suma += calificaciones[i];
    }

    // Calculo del promedio
    promedio = suma / 5;

    // Resultados
    printf("\nCalificaciones ingresadas:\n");
    for (int i = 0; i < 5; i++) {
        printf("  Materia %d: %.2f\n", i + 1, calificaciones[i]);
    }

    printf("\nPromedio: %.2f\n", promedio);

    if (promedio >= 60) {
        printf("Estado: APROBADO\n");
    } else {
        printf("Estado: REPROBADO\n");
    }

    return 0;
}

#Arreglo donde se guardan los datos
datos = []

# PILA

# Agrega al final
def push():
    valor = input("Ingresa un dato para la pila: ")
    datos.append(valor)


# Elimina el ultimo
def pop():
    if len(datos) == 0:
        print("No hay datos")
        return

    # .pop() elimina por defecto el ultimo dato si no se pone un indice
    eliminado = datos.pop()
    print("Dato eliminado:", eliminado)


# COLA

# Agrega al final
def enqueue():
    valor = input("Ingresa un dato para la cola: ")
    datos.append(valor)


# Elimina el primero
def dequeue():
    if len(datos) == 0:
        print("No hay datos")
        return

    # pop elimina el dato en el indice 0 del arreglo
    eliminado = datos.pop(0)
    print("Dato eliminado:", eliminado)


# Muestra la lista
def ver_arreglo():
    if len(datos) == 0:
        print("El arreglo esta vacio")
        return

    print("Arreglo:", datos)

    n = len(lista)

    lista.append(objetivo) # agregar centinela

    i = 0
    while lista[i] != objetivo:

# Bubble Sort ----------------------------------------------------
# Compara pares adyacentes y los intercambia si estan en orden incorrecto
n = len(my_array)
for i in range(n - 1):                      # pasadas necesarias
    for j in range(n - i - 1):              # evita comparar los ya ordenados al final
        if my_array[j] > my_array[j + 1]:  # elemento izquierdo es mayor
            my_array[j], my_array[j + 1] = my_array[j + 1], my_array[j]  # intercambio

print("Arreglo ordenado con Bubble Sort:", my_array)


# Selection Sort ----------------------------------------------------
# En cada pasada busca el minimo del subarreglo sin ordenar
# y lo mueve a la posicion correcta al inicio.
n = len(my_array)
for i in range(n - 1):          # posicion donde colocaremos el minimo
    min_index = i                # asumimos que el minimo esta en i
    for j in range(i + 1, n):   # buscar el minimo real en el resto
        if my_array[j] < my_array[min_index]:
            min_index = j        # actualizar indice del minimo encontrado
    min_value = my_array.pop(min_index)  # extraer el minimo
    my_array.insert(i, min_value)        # insertarlo en su posicion

print("Arreglo ordenado con Selection Sort:", my_array)


# Insertion Sort ----------------------------------------------------
# Toma cada elemento y lo inserta en el lugar correcto
# dentro de la parte izquierda que ya esta ordenada.
n = len(my_array)
for i in range(1, n):                       # recorre desde el segundo elemento
    insert_index = i
    current_value = my_array.pop(i)         # extrae el elemento a insertar
    for j in range(i - 1, -1, -1):         # recorre hacia atras la parte ordenada
        if my_array[j] > current_value:     # si el vecino es mayor, retrocede
            insert_index = j
    my_array.insert(insert_index, current_value)  # inserta en la posicion correcta

print("Arreglo ordenado con Insertion Sort:", my_array)


# Merge Sort ----------------------------------------------------
# Divide y Venceras
#  divide el arreglo a la mitad de forma
# recursiva hasta tener subarreglos de 1 elemento, luego las
# fusiona en orden.
def mergeSort(arr):
    if len(arr) <= 1:       # caso base: un elemento ya esta ordenado
        return arr

    mid = len(arr) // 2     # punto de division
    leftHalf  = arr[:mid]   # mitad izquierda
    rightHalf = arr[mid:]   # mitad derecha

    sortedLeft  = mergeSort(leftHalf)   # ordenar mitad izquierda (recursivo)
    sortedRight = mergeSort(rightHalf)  # ordenar mitad derecha (recursivo)

    return merge(sortedLeft, sortedRight)  # fusionar ambas mitades


def merge(left, right):
    result = []
    i = j = 0

    # Comparar elemento a elemento de cada mitad e insertar el menor
    while i < len(left) and j < len(right):
        if left[i] < right[j]:
            result.append(left[i])
            i += 1
        else:
            result.append(right[j])
            j += 1

    result.extend(left[i:])   # agregar elementos restantes de la izquierda
    result.extend(right[j:])  # agregar elementos restantes de la derecha

    return result


sortedArr = mergeSort(my_array)
print("Ordenado con Merge Sort:", sortedArr)

Meta 3.2 — Comparación de Algoritmos de Búsqueda

Programa que implementa y compara tres algoritmos de búsqueda (Secuencial, Binaria e Interpolada) midiendo su rendimiento en tiempo y número de operaciones sobre conjuntos de 100,000 datos ordenados y aleatorios, con 10,000 repeticiones por prueba para obtener mediciones estables.

algoritmos_busqueda_comparacion.py 

import random
import time

# ──────────────────────────────────────────────
#  DATOS
# ──────────────────────────────────────────────
datos_ordenados   = list(range(1, 100001))          # [1, 2, ..., 100000]
datos_aleatorios  = random.sample(range(1, 100001), 100000)

# Veces que se va a repetir la búsqueda (por algoritmo) para estabilizar la medición
REPETICIONES = 10000

# ──────────────────────────────────────────────
#  ALGORITMOS  (con contador de operaciones)
# ──────────────────────────────────────────────

def linearSearch(arr, x):
    ops = 0
    for i in range(len(arr)):
        ops += 1
        if arr[i] == x:
            return i, ops
    return -1, ops


def binarySearch(arr, x):
    ops  = 0
    low  = 0
    high = len(arr) - 1

    while low <= high:
        ops += 1
        mid  = low + (high - low) // 2

        if arr[mid] == x:
            return mid, ops
        elif arr[mid] < x:
            low = mid + 1
        else:
            high = mid - 1

    return -1, ops


def interpolationSearch(arr, x):
    ops = 0
    lo  = 0
    hi  = len(arr) - 1

    while lo <= hi and x >= arr[lo] and x <= arr[hi]:
        ops += 1

        if arr[hi] == arr[lo]:          # Evitar división entre cero
            if arr[lo] == x:
                return lo, ops
            break

        pos = lo + ((x - arr[lo]) * (hi - lo)) // (arr[hi] - arr[lo])

        if arr[pos] == x:
            return pos, ops
        elif arr[pos] < x:
            lo = pos + 1
        else:
            hi = pos - 1

    return -1, ops

# ──────────────────────────────────────────────
#  FUNCIÓN DE MEDICIÓN CORREGIDA
# ──────────────────────────────────────────────

def medir(funcion, arr, objetivo):
    idx, ops = -1, 0
    
    inicio = time.perf_counter()
    for _ in range(REPETICIONES):
        idx, ops = funcion(arr, objetivo)
    fin = time.perf_counter()
    
    tiempo_promedio = (fin - inicio) / REPETICIONES
    return idx, ops, tiempo_promedio

# ──────────────────────────────────────────────
#  FUNCIÓN DE PRUEBA
# ──────────────────────────────────────────────

def ejecutar_prueba(nombre_algoritmo, funcion, objetivo):
    print(f"\n{'─'*54}")
    print(f"  Algoritmo : {nombre_algoritmo}  ({REPETICIONES} repeticiones)")
    print(f"  Buscando  : {objetivo}")
    print(f"{'─'*54}")

    idx, ops, t_prom = medir(funcion, datos_ordenados, objetivo)
    resultado = f"índice {idx}" if idx != -1 else "no encontrado"
    print(f"  [Ordenados]   resultado={resultado:20s}  "
          f"ops={ops:6d}  t_prom={t_prom:.8f} s")

    if nombre_algoritmo == "Secuencial":
        arr_para_buscar = datos_aleatorios
        t_sort = 0.0
    else:
        t0 = time.perf_counter()
        arr_para_buscar = sorted(datos_aleatorios)
        t_sort = time.perf_counter() - t0

    idx, ops, t_prom = medir(funcion, arr_para_buscar, objetivo)
    resultado = f"índice {idx}" if idx != -1 else "no encontrado"
    extra = f"  (+ {t_sort:.6f} s de ordenamiento)" if t_sort > 0 else ""
    print(f"  [Aleatorios]  resultado={resultado:20s}  "
          f"ops={ops:6d}  t_prom={t_prom:.8f} s{extra}")

Meta 4.1 — Grafos y Ordenamiento Topológico

Sistema de gestión curricular que modela las dependencias entre materias universitarias mediante un Grafo Dirigido, implementando un algoritmo de Ordenamiento Topológico con DFS (Depth-First Search) para determinar el orden válido de cursado, incluyendo detección de dependencias circulares.

gestion_materias.py 

class Materia:
    """Representa una materia dentro del plan de estudios."""

    def __init__(self, nombre):
        self.nombre = nombre
        self.prerrequisitos_de = []  # materias que dependen de esta
        self.estado = 0  # 0: No visitado, 1: Visitando, 2: Procesado


class GrafoCurricular:
    """Grafo dirigido que modela las dependencias entre materias (prerrequisitos)."""

    def __init__(self):
        self.materias = {}  # nombre -> objeto Materia

    def agregar_materia(self, nombre):
        if nombre not in self.materias:
            self.materias[nombre] = Materia(nombre)

    def agregar_prerrequisito(self, materia_base, materia_dependiente):
        # Indica que 'materia_dependiente' requiere haber tomado 'materia_base'.
        # Esto crea una arista dirigida: materia_base -> materia_dependiente
        self.agregar_materia(materia_base)
        self.agregar_materia(materia_dependiente)
        self.materias[materia_base].prerrequisitos_de.append(materia_dependiente)

    def _dfs_visita(self, nombre_nodo, pila_resultado):
        nodo = self.materias[nombre_nodo]
        nodo.estado = 1  # Marcar como visitado

        for vecino in nodo.prerrequisitos_de:
            vecino_nodo = self.materias[vecino]
            
            # Deteccion de ciclos
            if vecino_nodo.estado == 1:
                raise ValueError(f"Error: Dependencia circular de materia: '{vecino}'. "
                                 f"El plan de estudios no puede resolverse.")
            
            elif vecino_nodo.estado == 0:
                self._dfs_visita(vecino, pila_resultado)

        nodo.estado = 2  # Marcamor como Procesado, sale de la recursion
        pila_resultado.append(nombre_nodo)

    def ordenamiento_topologico(self):
        # Regresa el orden en que se pueden cursar las materias.
        # Reiniciar estados por si se llama mas de una vez
        for materia in self.materias.values():
            materia.estado = 0

        pila_resultado = []

        for nombre_materia in self.materias:
            if self.materias[nombre_materia].estado == 0:
                self._dfs_visita(nombre_materia, pila_resultado)

        pila_resultado.reverse()
        return pila_resultado

Meta 4.2 — Árbol Binario de Búsqueda

Sistema de gestión de citas médicas implementado con un Árbol Binario de Búsqueda (BST), donde cada nodo representa una cita con hora, paciente y motivo. El árbol ordena las citas cronológicamente y detecta conflictos de horario cuando se intenta agendar dos citas a la misma hora.

algoritmo_arbol.py 

# Cita es Nodo del Arbol Binario de Busqueda
class Cita:
    def __init__(self, hora, paciente, motivo):
        self.hora = hora            # formato "00:00" (24h)
        self.paciente = paciente    # nombre del paciente
        self.motivo = motivo        # motivo de la consulta
        self.left = None            # cita anterior en el tiempo
        self.right = None           # cita posterior en el tiempo

# Funcion para insertar una nueva cita en el arbol de la agenda
def insertar_cita(root, nueva_cita):
    # Si la agenda esta vacia, la nueva cita se convierte en la raiz
    if root is None:
        return nueva_cita

    # Comparar la hora de la nueva cita con la cita actual (root)
    if nueva_cita.hora < root.hora:
        # Ir al subarbol izquierdo (citas mas temprano)
        root.left = insertar_cita(root.left, nueva_cita)
    elif nueva_cita.hora > root.hora:
        # Ir al subarbol derecho (citas mas tarde)
        root.right = insertar_cita(root.right, nueva_cita)
    else:
        # Si la hora es la misma, se detecta un conflicto de horario
        print(f"CONFLICTO: Ya existe una cita agendada a las {nueva_cita.hora} para {root.paciente}.")
        
    return root

# Recorrido Inorder para mostrar la agenda de forma cronologica
def mostrar_agenda(root):
    if root:
        # Visitar citas mas temprano (izquierda)
        mostrar_agenda(root.left)
        
        # Mostrar la cita actual
        print(f"| {root.hora:<7} | {root.paciente:<20} | {root.motivo:<25} |")
        
        # Visitar citas mas tarde (derecha)
        mostrar_agenda(root.right)

Meta 4.3 — Compresión de Huffman

Implementación completa del algoritmo de compresión de Huffman para archivos binarios. Incluye construcción del árbol de Huffman mediante un Min-Heap, generación de códigos de longitud variable, serialización con encabezado JSON, y un sistema de pruebas de rendimiento automatizadas que evalúa la compresión en cuatro escenarios distintos (texto redundante, texto literario, datos aleatorios y escalabilidad a 1 MB), validando la integridad mediante SHA-256.

huffman.py 

import heapq
import json
from collections import Counter

class HuffmanNode:
    """Clase que representa un nodo en el árbol de Huffman."""
    def __init__(self, byte_val, freq, left=None, right=None):
        self.byte_val = byte_val
        self.freq = freq
        self.left = left
        self.right = right

    def __lt__(self, other):
        """Permite que heapq ordene los nodos por frecuencia de menor a mayor."""
        return self.freq < other.freq

class HuffmanCompressor:
    """Clase para comprimir y descomprimir archivos usando el algoritmo de Huffman."""
    
    @staticmethod
    def _build_tree(freqs):
        """Construye el árbol de Huffman y devuelve el nodo raíz."""
        heap = [HuffmanNode(byte_val, freq) for byte_val, freq in freqs.items()]
        heapq.heapify(heap)

        if len(heap) == 1:
            node = heapq.heappop(heap)
            parent = HuffmanNode(None, node.freq, left=node)
            heapq.heappush(heap, parent)

        while len(heap) > 1:
            node1 = heapq.heappop(heap)
            node2 = heapq.heappop(heap)
            merged = HuffmanNode(None, node1.freq + node2.freq, left=node1, right=node2)
            heapq.heappush(heap, merged)

        return heap[0] if heap else None

    @staticmethod
    def _generate_codes(root):
        """Genera el diccionario de códigos de Huffman (byte -> secuencia de bits)."""
        codes = {}
        
        def traverse(node, current_code):
            if node is None:
                return
            if node.byte_val is not None:
                codes[node.byte_val] = current_code
                return
            traverse(node.left, current_code + '0')
            traverse(node.right, current_code + '1')

        traverse(root, "")
        return codes

    def compress(self, input_path, output_path):
        """Comprime un archivo de entrada y guarda el resultado en el archivo de salida."""
        with open(input_path, 'rb') as f:
            data = f.read()

        if not data:
            with open(output_path, 'wb') as f_out:
                f_out.write((0).to_bytes(4, 'big'))
                f_out.write((0).to_bytes(1, 'big'))
            return

        freqs = dict(Counter(data))
        root = self._build_tree(freqs)
        codes = self._generate_codes(root)
        bit_string = "".join(codes[byte] for byte in data)
        padding = (8 - len(bit_string) % 8) % 8
        bit_string += "0" * padding

        byte_arr = bytearray()
        for i in range(0, len(bit_string), 8):
            byte_arr.append(int(bit_string[i:i+8], 2))

        json_freqs = json.dumps({str(k): v for k, v in freqs.items()}).encode('utf-8')
        header_len = len(json_freqs)

        with open(output_path, 'wb') as f_out:
            f_out.write(header_len.to_bytes(4, 'big'))
            f_out.write(json_freqs)
            f_out.write(padding.to_bytes(1, 'big'))
            f_out.write(byte_arr)

    def decompress(self, input_path, output_path):
        """Descomprime un archivo comprimido de Huffman y recupera el archivo original."""
        with open(input_path, 'rb') as f:
            header_len_bytes = f.read(4)
            if not header_len_bytes:
                with open(output_path, 'wb') as f_out:
                    f_out.write(b"")
                return
            
            header_len = int.from_bytes(header_len_bytes, 'big')
            if header_len == 0:
                with open(output_path, 'wb') as f_out:
                    f_out.write(b"")
                return

            json_freqs = json.loads(f.read(header_len).decode('utf-8'))
            freqs = {int(k): v for k, v in json_freqs.items()}
            padding = int.from_bytes(f.read(1), 'big')
            compressed_data = f.read()

        bit_string = "".join(f"{b:08b}" for b in compressed_data)
        
        if padding > 0:
            bit_string = bit_string[:-padding]

        root = self._build_tree(freqs)
        decoded_bytes = bytearray()
        current_node = root
        for bit in bit_string:
            if bit == '0':
                current_node = current_node.left
            else:
                current_node = current_node.right

            if current_node.left is None and current_node.right is None:
                decoded_bytes.append(current_node.byte_val)
                current_node = root

        with open(output_path, 'wb') as f_out:
            f_out.write(decoded_bytes)

def calculate_sha256(filepath):
    """Calcula el hash SHA-256 de un archivo para verificar integridad."""
    sha256 = hashlib.sha256()
    with open(filepath, 'rb') as f:
        while chunk := f.read(8192):
            sha256.update(chunk)
    return sha256.hexdigest()

Meta 5.1 — Árbol Trie y Optimización de Recursos

Implementación optimizada de un Árbol de Prefijos (Trie) con autocompletado por relevancia mediante Min-Heap, seguridad ante concurrencia con RLock, y reducción de overhead de memoria mediante __slots__. Incluye simulación de benchmark comparativo entre Trie y Lista convencional sobre un diccionario de 1,100 palabras generadas combinatoriamente.

trie.py 

import threading
import sys
import heapq
from typing import Dict, Any, List, Tuple, Optional

class TrieNode:
    """
    Nodo del arbol Trie.
    Usa __slots__ para reducir consumo de memoria RAM.
    """
    __slots__ = ('children', 'is_end_of_word', 'frequency', 'value')

    def __init__(self):
        self.children: Dict[str, 'TrieNode'] = {}
        self.is_end_of_word: bool = False
        self.frequency: int = 0
        self.value: Any = None

class Trie:
    """
    Arbol Trie optimizado con Slots, Thread-safety y heapq.
    """
    __slots__ = ('root', '_lock')

    def __init__(self):
        self.root = TrieNode()
        self._lock = threading.RLock()

    def insert(self, word: str, value: Any = None, frequency: int = 1) -> None:
        if not word:
            return

        word = word.lower()
        with self._lock:
            current = self.root
            for char in word:
                if char not in current.children:
                    current.children[char] = TrieNode()
                current = current.children[char]
            
            current.is_end_of_word = True
            current.frequency += frequency
            if value is not None:
                current.value = value

    def search(self, word: str) -> Tuple[bool, Optional[Any], int]:
        if not word:
            return False, None, 0

        word = word.lower()
        with self._lock:
            current = self.root
            for char in word:
                if char not in current.children:
                    return False, None, 0
                current = current.children[char]
            
            if current.is_end_of_word:
                return True, current.value, current.frequency
            return False, None, 0

    def delete(self, word: str) -> bool:
        if not word:
            return False

        word = word.lower()
        with self._lock:
            success, _ = self._delete_helper(self.root, word, 0)
            return success

    def _delete_helper(self, current: TrieNode, word: str, index: int) -> Tuple[bool, bool]:
        if index == len(word):
            if not current.is_end_of_word:
                return False, False
            current.is_end_of_word = False
            current.frequency = 0
            current.value = None
            return True, len(current.children) == 0

        char = word[index]
        if char not in current.children:
            return False, False

        success, should_delete_child = self._delete_helper(current.children[char], word, index + 1)

        if should_delete_child:
            del current.children[char]
            return success, len(current.children) == 0 and not current.is_end_of_word

        return success, False

    def autocomplete(self, prefix: str, limit: int = 5) -> List[Tuple[str, int, Any]]:
        prefix = prefix.lower()
        with self._lock:
            current = self.root
            for char in prefix:
                if char not in current.children:
                    return []
                current = current.children[char]

            heap: List[Tuple[int, str, Any]] = []
            
            def dfs(node: TrieNode, current_word: str):
                if node.is_end_of_word:
                    item = (node.frequency, current_word, node.value)
                    if len(heap) < limit:
                        heapq.heappush(heap, item)
                    else:
                        if node.frequency > heap[0][0]:
                            heapq.heappushpop(heap, item)

                for char, child_node in node.children.items():
                    dfs(child_node, current_word + char)

            dfs(current, prefix)

            results = []
            while heap:
                freq, word, val = heapq.heappop(heap)
                results.append((word, freq, val))
            
            return results[::-1]

    def __setitem__(self, key: str, value: Any) -> None:
        self.insert(key, value, frequency=1)

    def __getitem__(self, key: str) -> Any:
        exists, value, _ = self.search(key)
        if not exists:
            raise KeyError(key)
        return value

    def __contains__(self, key: str) -> bool:
        exists, _, _ = self.search(key)
        return exists

    def get(self, key: str, default: Any = None) -> Any:
        exists, value, _ = self.search(key)
        return value if exists else default

    def get_stats(self) -> Dict[str, Any]:
        total_nodes = 0
        total_words = 0
        total_characters_in_words = 0

        with self._lock:
            def traverse(node: TrieNode, depth: int):
                nonlocal total_nodes, total_words, total_characters_in_words
                total_nodes += 1
                if node.is_end_of_word:
                    total_words += 1
                    total_characters_in_words += depth
                
                for child in node.children.values():
                    traverse(child, depth + 1)

            traverse(self.root, 0)

        trie_mem = self._estimate_trie_memory(self.root)
        list_est_mem = 0
        words_list = []
        
        with self._lock:
            def collect_words(node: TrieNode, current_word: str):
                if node.is_end_of_word:
                    words_list.append(current_word)
                for char, child in node.children.items():
                    collect_words(child, current_word + char)
            collect_words(self.root, "")

        for w in words_list:
            list_est_mem += sys.getsizeof(w)
        list_est_mem += sys.getsizeof(words_list)

        ahorro_bytes = max(0, list_est_mem - trie_mem)
        porcentaje_ahorro = (ahorro_bytes / list_est_mem * 100) if list_est_mem > 0 else 0

        return {
            "total_nodos": total_nodes,
            "total_palabras": total_words,
            "caracteres_totales_sin_trie": total_characters_in_words,
            "memoria_trie_estimada_bytes": trie_mem,
            "memoria_lista_estimada_bytes": list_est_mem,
            "ahorro_memoria_bytes": ahorro_bytes,
            "porcentaje_ahorro_memoria": round(porcentaje_ahorro, 2)
        }

    def _estimate_trie_memory(self, node: TrieNode) -> int:
        size = sys.getsizeof(node)
        size += sys.getsizeof(node.children)
        
        for char in node.children.keys():
            size += sys.getsizeof(char)
        
        for child in node.children.values():
            size += self._estimate_trie_memory(child)
            
        return size

Reflexión Final: Importancia de la Optimización de Recursos de la Computadora

A lo largo de esta asignatura, he consolidado un entendimiento riguroso sobre cómo la selección adecuada de estructuras de datos y algoritmos incide directamente en la eficiencia y escalabilidad de los sistemas de software. Desde la implementación de arreglos estáticos y la medición del costo computacional de operaciones aritméticas con distintos tipos de dato en la Unidad I, hasta el desarrollo de estructuras avanzadas como Árboles de Prefijos (Tries) optimizados con __slots__ y Min-Heaps en la Unidad V, cada práctica me ha permitido observar de primera mano la diferencia entre una solución funcional y una solución eficiente.

La optimización de recursos de la computadora —tanto en tiempo de CPU como en consumo de memoria RAM— no es un lujo académico, sino una necesidad imperativa en el desarrollo de software profesional. Algoritmos con complejidad cuadrática O(n²), como Bubble Sort, resultan funcionales para conjuntos pequeños de datos, pero escalan de manera inaceptable conforme el volumen de información crece. Refactorizar hacia enfoques de complejidad O(n log n), como Merge Sort, o incluso O(log n), como la Búsqueda Binaria, representa la diferencia entre un sistema que responde en milisegundos y uno que colapsa bajo carga real.

El análisis exhaustivo de los tres escenarios posibles —mejor, promedio y peor caso— para cada algoritmo implementado me ha brindado un criterio de ingeniería fundamental: anticipar el comportamiento del software bajo condiciones extremas antes de desplegarlo en producción. Este rigor analítico, combinado con la implementación práctica de estructuras no lineales como Grafos Dirigidos con Ordenamiento Topológico, Árboles Binarios de Búsqueda, algoritmos de compresión como Huffman y estructuras de indexación como el Trie, me ha dotado de las herramientas necesarias para diseñar soluciones que no solo funcionen correctamente, sino que brinden la mejor experiencia de usuario posible con un consumo óptimo de los recursos computacionales disponibles.

En conclusión, la ingeniería de software trasciende la mera escritura de código funcional. Requiere un dominio profundo de la teoría algorítmica, una comprensión clara de las limitaciones del hardware y del lenguaje de programación utilizado, y la capacidad de evaluar trade-offs entre tiempo, espacio y complejidad de implementación. Las competencias adquiridas en esta asignatura constituyen un pilar fundamental para mi formación como desarrollador de software, permitiéndome abordar cualquier problemática computacional con criterio técnico, eficiencia y responsabilidad profesional.
